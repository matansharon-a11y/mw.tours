-- =====================================================================
-- MW Photo Tours · update: change an expedition's code from the admin page
-- Supabase → SQL Editor → New query → paste this whole file → Run. Safe to run more than once.
--
-- The code travelers type and the key the page is locked with are kept apart (join_code_hash / content_secret),
-- so a new code needs no change to the page. The old code stops working; travelers who already joined stay in.
-- join_code keeps the code as Matan typed it, so the admin page can show it (tours has no policies: not readable
-- from outside, only through the admin functions below).
-- =====================================================================

alter table public.tours add column if not exists join_code text;

-- connect an expedition / change its code (p_secret only the first time; p_code alone = a new code)
create or replace function public.admin_set_tour(p_slug text, p_title text, p_secret text, p_code text) returns void
language plpgsql security definer set search_path = public, extensions as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  if p_code is not null and char_length(public.norm_code(p_code)) < 4 then raise exception 'code_too_short'; end if;
  insert into public.tours (slug, title, content_secret, join_code_hash, join_code)
  values (p_slug, p_title, nullif(public.norm_code(p_secret), ''),
          case when p_code is null then null else extensions.crypt(public.norm_code(p_code), extensions.gen_salt('bf')) end,
          case when p_code is null then null else upper(regexp_replace(btrim(p_code), '\s+', '', 'g')) end)
  on conflict (slug) do update set
    title          = coalesce(excluded.title, public.tours.title),
    content_secret = coalesce(excluded.content_secret, public.tours.content_secret),
    join_code_hash = coalesce(excluded.join_code_hash, public.tours.join_code_hash),
    join_code      = coalesce(excluded.join_code, public.tours.join_code),
    updated_at     = now();
end $$;

-- the admin page: same as before, the code now comes from join_code (older rows: the page's own code), plus code_edit
create or replace function public.admin_overview() returns json
language plpgsql stable security definer set search_path = public, extensions as $$
begin
  if not public.is_tour_admin() then raise exception 'not_admin'; end if;
  return coalesce((
    select json_agg(json_build_object(
      'slug', t.slug, 'title', t.title, 'ready', t.join_code_hash is not null and t.content_secret is not null, 'updated_at', t.updated_at,
      'code', coalesce(t.join_code, case when t.join_code_hash is not null and extensions.crypt(t.content_secret, t.join_code_hash) = t.join_code_hash then t.content_secret end),
      'code_edit', true,
      'travelers', coalesce((select json_agg(json_build_object('id', r.id, 'name', r.name, 'other_names', r.other_names) order by lower(r.name))
        from public.tour_travelers r where r.tour_slug = t.slug), '[]'::json),
      'members', coalesce((select json_agg(json_build_object(
          'id', m.id, 'name', m.name, 'traveler_id', m.traveler_id, 'joined', m.created_at, 'last_seen', m.last_seen, 'visits', m.visits,
          'photos', (select count(*) from public.photos p where p.member_id = m.id)) order by m.last_seen desc)
        from public.members m where m.tour_slug = t.slug), '[]'::json)
    ) order by t.slug) from public.tours t), '[]'::json);
end $$;

grant execute on function public.admin_set_tour(text, text, text, text), public.admin_overview() to authenticated;
